### 3. `.github/workflows/snake.yml`

```yaml
name: Generate Snake Animation

on:
  schedule:
    - cron: "0 0 * * *"

  workflow_dispatch:

  push:
    branches:
      - main

jobs:
  generate:
    permissions:
      contents: write

    runs-on: ubuntu-latest

    steps:

      - name: Generate contribution snake
        uses: Platane/snk@v3
        with:
          github_user_name: Paulohenrique200377

          outputs: |
            dist/github-contribution-grid-snake.svg
            dist/github-contribution-grid-snake-dark.svg?palette=github-dark&color_snake=#00F7FF

      - name: Publish Snake
        uses: crazy-max/ghaction-github-pages@v4
        with:
          build_dir: dist
          branch: output

        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
```
